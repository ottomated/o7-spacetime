<h1 align="center">@o7/spacetime</h1>

<p align="center">Actually good Svelte wrapper for <a href="https://spacetimedb.com">SpacetimeDB</a>.</p>
<br />

## Basic Usage

```svelte
<!-- src/routes/+layout.svelte -->
<script lang="ts">
  import { createSpacetime } from '@o7/spacetime';
  import { DbConnection } from './bindings'; // Generated by `spacetime generate`

  let { children } = $props();

  createSpacetime(
    DbConnection.builder()
      .withUri('http://localhost:3000')
      .withDatabaseName('my-db'),
  );
</script>

{@render children()}
```

```svelte
<!-- src/routes/+page.svelte -->
<script lang="ts">
  import { reducer, table } from '@o7/spacetime';
  import { reducers, tables } from './bindings';

  const messages = table(tables.message);

  const send = reducer(reducers.sendMessage);

  let message = $state.raw('');
</script>

<input bind:value={message} />
<button onclick={() => send({ message })}>send</button>

{#each messages.rows as msg (msg.id)}
  {msg.author}: {msg.message}
{/each}
```

## Dynamic token

```svelte
<!-- src/routes/+layout.svelte -->
<script lang="ts">
  import { createSpacetime } from '@o7/spacetime';
  import { DbConnection } from './bindings';

  const { data } = $props();

  createSpacetime(() =>
    DbConnection.builder()
      .withUri('http://localhost:3000')
      .withToken(data.token)
      .withDatabaseName('my-db'),
  );
</script>
```
